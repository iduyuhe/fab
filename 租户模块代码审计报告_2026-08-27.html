T1 · tenant/index.js attachToEv

// 修复后：多租户下从事件中显式携带的 tenant 提取并附到 ev.tenant
attachToEv(ev) {
  if (!ev || typeof ev !== 'object') return ev;
  if (!MULTI_TENANT) return ev;            // 单租户：原样返回（不新增字段）
  ev.tenant = getTenant(ev);               // 改用已 import 的 getTenant，真正透传
  return ev;
}
// getTenant 提取源：ev.tenant / ev.meta.tenant / ev.headers['x-tenant-id']，缺省回退 DEFAULT_TENANT
